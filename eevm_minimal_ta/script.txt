char* strtok(char* s, const char* delim) {
    static char* last = (char*)0;
    if (s) last = s;
    if (!last) return (char*)0;
    
    // Skip leading delimiters
    while (*last) {
        const char* d = delim;
        while (*d && *d != *last) d++;
        if (!*d) break;
        last++;
    }
    
    if (!*last) return (char*)0;
    
    char* token = last;
    while (*last) {
        const char* d = delim;
        while (*d && *d != *last) d++;
        if (*d) {
            *last++ = 0;
            return token;
        }
        last++;
    }
